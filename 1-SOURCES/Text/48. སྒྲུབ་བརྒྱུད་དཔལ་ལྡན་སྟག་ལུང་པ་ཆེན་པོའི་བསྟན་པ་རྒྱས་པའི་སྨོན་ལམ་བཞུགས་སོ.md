# ༄༅། །སྒྲུབ་བརྒྱུད་དཔལ་ལྡན་སྟག་ལུང་པ་ཆེན་པོའི་བསྟན་པ་རྒྱས་པའི་སྨོན་ལམ་བཞུགས་སོ།། ^0

## སྒྲུབ་བརྒྱུད་དཔལ་ལྡན་སྟག་ལུང་པ་ཆེན་པོའི་བསྟན་པ་རྒྱས་པའི་སྨོན་ལམ། ^1-0

བཀྲ་ཤིས་ཤེས་བྱའི་ཁྱོན་འདིར་སྲིད་དང་ཞི། །
དཔལ་ཡོན་ཇི་སྙེད་གང་གི་ཞབས་པད་ཟུང་། །
བསྟེན་པས་བདེ་བླག་སྦྱིན་མཛད་སྟག་ལུང་པ། །
མི་འཁྲུགས་རྒྱལ་བའི་དབང་པོར་མགོས་ཕྱག་འཚལ། ། ^1-1

དགྱེས་མཛད་རྡོ་རྗེ་ཚུལ་ཁྲིམས་རྒྱན་མཛེས་ཞེས། །
གྲགས་པ་དོན་ལྡན་ནང་གི་མུན་འཇོམས་དཔལ། །
གྲུབ་བརྙེས་འོད་ཟེར་བསྐལ་བཟང་རྒྱ་ཆེན་པོར། །
དབུགས་དབྱུང་སངས་རྒྱས་དབོན་ལ་གསོལ་བ་འདེབས། ། ^1-2

མཁའ་ཁྱབ་ཞིང་བཞུགས་རྒྱལ་དང་དེས་གསུང་ཆོས། །
ས་བཅུའི་རྒྱལ་སྲས་ཉན་རང་འདུལ་བའི་ཆོས། །
རྩ་བ་གསུམ་སོགས་སྐྱབས་གནས་བདེན་གྲུབ་ཀྱི། །
ཚད་མར་གྱུར་ཀུན་སྨོན་གསོལ་དཔང་པོར་སྤྱོན། ། ^1-3

ཐུབ་གཞུང་ཇི་སྙེད་སྒྲ་ལས་དེ་ཁོ་ནའི། །
ཡེ་ཤེས་མངོན་གསུམ་སྟོན་ལ་དྭགས་རྒྱུད་ཅེས། །
མོས་གུས་མཐར་ཕྱིན་དཔལ་ལྡན་སྟག་ལུང་པའི། །
བསྟན་པ་རབ་རྒྱས་དགེ་འདུན་སྡེ་འཕེལ་ཤོག ། ^1-4

སྲིད་པའི་ཉེས་སྡོམ་རྣམ་པར་གྲོལ་བའི་སྒོ། །
ཚུལ་ཁྲིམས་ཏིང་འཛིན་ཤེས་རབ་བསླབ་གསུམ་གྱི། །
ཚངས་སྤྱོད་དྲི་མེད་དཔལ་ལྡན་སྟག་ལུང་པའི། །
བསྟན་པ་རབ་རྒྱས་དགེ་འདུན་སྡེ་འཕེལ་ཤོག ། ^1-5

ཞི་བདེ་སྲི་འགོག་རྣམ་མཁྱེན་རྒྱུའི་ས་བོན། །
ཕྱིན་དྲུག་སྤྱོད་པ་རྒྱ་མཚོའི་གཞན་དོན་འཇུག །
རྒྱལ་སྲས་མཆོག་གྱུར་དཔལ་ལྡན་སྟག་ལུང་པའི། །
བསྟན་པ་རབ་རྒྱས་དགེ་འདུན་སྡེ་འཕེལ་ཤོག ། ^1-6

སྣོད་བཅུད་སྣང་གྲགས་རྟོག་བཅས་ཡེ་ཤེས་ཀྱི། །
འཁོར་ལོ་པྲ་ཕབ་གཟུགས་བཞིར་རང་ཤར་ལྷ། །
རིག་པ་འཛིན་མཆོག་དཔལ་ལྡན་སྟག་ལུང་པའི། །
བསྟན་པ་རབ་རྒྱས་དགེ་འདུན་སྡེ་འཕེལ་ཤོག ། ^1-7

བསླབ་གསུམ་བྱང་ཆུབ་སྤྱོད་པའི་གོས་ཀྱིས་མཛེས། །
ཟབ་དོན་ནོར་བུ་རྣམ་བཞིའི་བང་མཛོད་མཆོག །
རྣམ་ཐར་མཚུངས་མེད་དཔལ་ལྡན་སྟག་ལུང་པའི། །
བསྟན་པ་རབ་རྒྱས་དགེ་འདུན་སྡེ་འཕེལ་ཤོག ། ^1-8

བག་ཡོད་ཁྲིམས་ཀྱི་རྒྱུད་བསྡམས་རང་ཆོས་འཛིན། །
མང་དུ་ཐོས་ཤིང་དུལ་སེམས་ཡང་དག་འཇོག །
རྣམ་དག་ངུར་སྨྲིག་གོས་ཅན་སཾ་གྷ་རྩེན། །
གྲངས་མེད་འཕེལ་རྒྱས་བསྟན་པའི་མཛེས་བྱེད་ཤོག ། ^1-9

འཇིག་རྟེན་གསུམ་ན་ཕན་བདེ་མ་ལུས་པའི། །
འབྱུང་གནས་སྡུག་བསྔལ་རྒྱུ་བཅས་ཞི་བའི་སྨན། །
ཐུབ་བསྟན་སྤྱི་དང་གྲུབ་རིགས་སྟག་ལུང་པའི། །
རིང་ལུགས་སྲིད་པའི་རྩེ་མོ་འབར་གྱུར་ཅིག ། ^1-10

མ་ལུས་དཀོན་མཆོག་རྒྱ་མཚོའི་ཐུགས་རྗེའི་བྱིན། །
བདེན་པར་སྨྲ་བ་རྣམས་ཀྱི་ཚིག་གི་མཐུས། །
རྡོ་རྗེ་དམ་ཅན་སྲུང་མ་རྣོ་མྱུར་གྱིས། །
ཇི་ལྟར་སྨོན་པ་མཐའ་དག་འགྲུབ་གྱུར་ཅིག ། ^1-11

<small>ཅེས་པ་འདི་ཡང་སྟག་ལུང་པ་ཆོས་དབྱིངས་ལྷུན་གྲུབ་ནས་བྲིས་པ་དགེའོ།།</small> ^1-12
